Supabase schema

Run these statements in order against a new Supabase project (SQL Editor, or as a migration). They create the content tables, the role model, revision history, and row-level security policies for the admin panel.

1. Enumerated types

create type content_status as enum ('draft', 'in_review', 'published', 'archived');
create type user_role     as enum ('editor', 'admin');
create type board_body    as enum ('directors', 'advisors');

2. Profiles and roles

Extends Supabase’s built-in auth.users. Accounts are created by invitation from the Supabase dashboard or the admin panel; there is no public sign-up.

create table profiles (
  id          uuid primary key references auth.users on delete cascade,
  email       text not null,
  full_name   text,
  role        user_role not null default 'editor',
  created_at  timestamptz not null default now()
);

create or replace function is_admin() returns boolean
language sql security definer stable as $$
  select exists (
    select 1 from profiles
    where id = auth.uid() and role = 'admin'
  );
$$;

create or replace function is_staff() returns boolean
language sql security definer stable as $$
  select exists (select 1 from profiles where id = auth.uid());
$$;

3. Board members

create table board_members (
  id            uuid primary key default gen_random_uuid(),
  body          board_body not null,
  full_name     text not null,
  position      text not null,
  organization  text,
  sector        text,
  email         text,
  linkedin_url  text,
  bio           text,
  headshot_path text,
  sort_order    int not null default 0,
  status        content_status not null default 'draft',
  updated_by    uuid references profiles(id),
  updated_at    timestamptz not null default now()
);

create index on board_members (body, sort_order);

4. Articles and newsletters

Bodies are stored as structured JSON blocks, not HTML, so the public site renders them in the chapter’s own type styles.

create table articles (
  id           uuid primary key default gen_random_uuid(),
  slug         text unique not null,
  title        text not null,
  standfirst   text,
  body         jsonb not null default '[]'::jsonb,
  author_id    uuid references board_members(id),
  category     text,
  read_minutes int,
  hero_path    text,
  featured     boolean not null default false,
  status       content_status not null default 'draft',
  published_at timestamptz,
  updated_by   uuid references profiles(id),
  updated_at   timestamptz not null default now()
);

create table newsletters (
  id            uuid primary key default gen_random_uuid(),
  issue_number  int unique,
  subject       text not null,
  body          jsonb not null default '[]'::jsonb,
  article_ids   uuid[] default '{}',
  sent_at       timestamptz,
  status        content_status not null default 'draft',
  updated_by    uuid references profiles(id),
  updated_at    timestamptz not null default now()
);

6. Revision history

One table covers every content type. The trigger snapshots a row on each update, so any version can be restored.

create table revisions (
  id          bigserial primary key,
  table_name  text not null,
  record_id   uuid not null,
  snapshot    jsonb not null,
  changed_by  uuid references profiles(id),
  changed_at  timestamptz not null default now()
);

create index on revisions (table_name, record_id, changed_at desc);

create or replace function record_revision() returns trigger
language plpgsql security definer as $$
begin
  insert into revisions (table_name, record_id, snapshot, changed_by)
  values (tg_table_name, old.id, to_jsonb(old), auth.uid());
  return new;
end;
$$;

create trigger rev_articles      before update on articles
  for each row execute function record_revision();
create trigger rev_newsletters   before update on newsletters
  for each row execute function record_revision();
create trigger rev_board_members before update on board_members
  for each row execute function record_revision();
create trigger rev_events        before update on events
  for each row execute function record_revision();

7. Row-level security

The public reads only published rows. Signed-in staff read everything and may write drafts. Publishing, deleting, and user management are restricted to administrators. Apply the same three-policy pattern to each content table.

alter table board_members enable row level security;
alter table articles      enable row level security;
alter table newsletters   enable row level security;
alter table events        enable row level security;
alter table sponsors      enable row level security;
alter table site_copy     enable row level security;
alter table profiles      enable row level security;
alter table revisions     enable row level security;

-- pattern, repeated per content table
create policy public_read on articles
  for select using (status = 'published');

create policy staff_read on articles
  for select using (is_staff());

create policy staff_write on articles
  for insert with check (is_staff());

create policy staff_update on articles
  for update using (is_staff())
  with check (is_admin() or status <> 'published');

create policy admin_delete on articles
  for delete using (is_admin());

-- profiles: read own row; only admins manage roles
create policy own_profile on profiles
  for select using (id = auth.uid() or is_admin());
create policy admin_manage_profiles on profiles
  for all using (is_admin()) with check (is_admin());

8. Storage buckets

insert into storage.buckets (id, name, public)
values ('headshots', 'headshots', true),
       ('media',     'media',     true),
       ('logos',     'logos',     true);

create policy staff_upload on storage.objects
  for insert to authenticated
  with check (bucket_id in ('headshots','media','logos') and is_staff());

create policy public_view on storage.objects
  for select using (bucket_id in ('headshots','media','logos'));
